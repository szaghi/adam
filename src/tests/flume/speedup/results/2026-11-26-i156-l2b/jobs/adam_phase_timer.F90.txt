!< Measurement-only per-phase wall-clock timers of a FLUME time step (issue #133, night 2026-11-24): NOT for commit.
module adam_phase_timer
!< Cumulative MPI_Wtime timers per phase of the step, armed by ADAM_PHASE_TIMING=1; the first ADAM_PHASE_SKIP steps
!< (default 10) are discarded; ADAM_FTZ=1 sets abrupt underflow in the forest's time loop; ADAM_GHOST_BARRIER=0 skips the two MPI_Barrier of every MPI ghost exchange. The report
!< gives, per phase, the time per step (min, mean, max over the ranks), the calls per step and the share of the step.
!< FNL kernels are synchronous (no async clause anywhere), so a host timer around a call measures its device time.
use :: penf, only : I4P, R8P
use :: mpi
use, intrinsic :: ieee_arithmetic, only : ieee_get_underflow_mode

implicit none
private
public :: PH_STEP, PH_DT_LOCAL, PH_DT_ALLREDUCE, PH_IB, PH_GHOST_LOCAL, PH_GHOST_MPI, PH_MPI_PACK, PH_MPI_POST, &
          PH_MPI_WAITALL, PH_MPI_BARRIER1, PH_MPI_UNPACK, PH_MPI_BARRIER2, PH_BC, PH_Q_AUX, PH_FACE_FLUXES,      &
          PH_FLUX_DIFF, PH_INTEGRATOR, PH_RESIDUALS_SAVE, PH_POST_STEP, PH_REGRID_DONE, PH_PROGRESS, PH_EVOLVE
public :: ph_on, ph_skip_barrier, ph_ftz, ph_init, ph_tic, ph_toc, ph_end_step, ph_report

integer(I4P), parameter :: PH_STEP           = 1  !< The loop iteration: evolve, regrid, post step, is_done.
integer(I4P), parameter :: PH_DT_LOCAL       = 2  !< Local dt (lambda max).
integer(I4P), parameter :: PH_DT_ALLREDUCE   = 3  !< The dt MPI_ALLREDUCE.
integer(I4P), parameter :: PH_IB             = 4  !< Immersed boundary: eikonal extension and inversion kernels.
integer(I4P), parameter :: PH_GHOST_LOCAL    = 5  !< Local (intra-rank) ghost fill.
integer(I4P), parameter :: PH_GHOST_MPI      = 6  !< MPI ghost exchange, whole (pack .. barrier 2).
integer(I4P), parameter :: PH_MPI_PACK       = 7  !< ... pack of the send buffer.
integer(I4P), parameter :: PH_MPI_POST       = 8  !< ... MPI_IRECV and MPI_ISEND posting.
integer(I4P), parameter :: PH_MPI_WAITALL    = 9  !< ... MPI_WAITALL.
integer(I4P), parameter :: PH_MPI_BARRIER1   = 10 !< ... the MPI_Barrier after MPI_WAITALL.
integer(I4P), parameter :: PH_MPI_UNPACK     = 11 !< ... unpack of the receive buffer.
integer(I4P), parameter :: PH_MPI_BARRIER2   = 12 !< ... the closing MPI_Barrier.
integer(I4P), parameter :: PH_BC             = 13 !< Boundary conditions and the ghost positivity blend.
integer(I4P), parameter :: PH_Q_AUX          = 14 !< Auxiliary (primitive) variables.
integer(I4P), parameter :: PH_FACE_FLUXES    = 15 !< Face fluxes (WENO reconstruction and flux), every direction.
integer(I4P), parameter :: PH_FLUX_DIFF      = 16 !< Flux difference (the residual).
integer(I4P), parameter :: PH_INTEGRATOR     = 17 !< Integrator: begin/end step and stage (SELAH), set_rhs on FNL.
integer(I4P), parameter :: PH_RESIDUALS_SAVE = 18 !< save_residuals (its MPI_ALLREDUCE on its cadence).
integer(I4P), parameter :: PH_POST_STEP      = 19 !< The forest's post step (non-finite check and its MPI_ALLREDUCE, ...).
integer(I4P), parameter :: PH_REGRID_DONE    = 20 !< The forest's regrid and is_done (two MPI_ALLREDUCE).
integer(I4P), parameter :: PH_PROGRESS       = 21 !< print_progress.
integer(I4P), parameter :: PH_EVOLVE         = 22 !< evolve_one_step (dt and the realm's step).
integer(I4P), parameter :: PH_N              = 22 !< Number of phases.

character(len=14), parameter :: PH_NAME(PH_N) = [character(len=14) :: 'step (loop)', 'dt local', 'dt allreduce',     &
   'IB kernels', 'ghost local', 'ghost MPI', '  pack', '  post', '  waitall', '  barrier 1', '  unpack', '  barrier 2', &
   'BC + blend', 'q_aux', 'face fluxes', 'flux diff', 'integrator', 'residuals save', 'post step', 'regrid+is_done',  &
   'progress', 'evolve']

logical      :: ph_on = .false.           !< Timers armed.
logical      :: ph_skip_barrier = .false. !< Skip the two barriers of the MPI ghost exchange.
logical      :: ph_ftz = .false.          !< Abrupt underflow (flush to zero) in the time loop, ADAM_FTZ=1.
real(R8P)    :: ph_acc(PH_N) = 0._R8P     !< Accumulated seconds.
real(R8P)    :: ph_t0(PH_N) = 0._R8P      !< Start stamps.
integer(I4P) :: ph_cnt(PH_N) = 0_I4P      !< Calls.
integer(I4P) :: ph_skip = 10_I4P          !< Steps discarded before accumulating.
integer(I4P) :: ph_it = 0_I4P             !< Steps done.
integer(I4P) :: ph_steps = 0_I4P          !< Steps accumulated.

contains
   subroutine ph_init()
   !< Read the environment: ADAM_PHASE_TIMING, ADAM_PHASE_SKIP, ADAM_GHOST_BARRIER.
   character(len=16) :: val  !< Environment value.
   integer           :: stat !< Query status.

   call get_environment_variable('ADAM_PHASE_TIMING', value=val, status=stat)
   ph_on = (stat == 0 .and. trim(val) == '1')
   call get_environment_variable('ADAM_PHASE_SKIP', value=val, status=stat)
   if (stat == 0) read(val, *) ph_skip
   call get_environment_variable('ADAM_GHOST_BARRIER', value=val, status=stat)
   ph_skip_barrier = (stat == 0 .and. trim(val) == '0')
   call get_environment_variable('ADAM_FTZ', value=val, status=stat)
   ph_ftz = (stat == 0 .and. trim(val) == '1')
   endsubroutine ph_init

   subroutine ph_tic(i)
   !< Start phase i.
   integer(I4P), intent(in) :: i !< Phase.

   if (ph_on) ph_t0(i) = MPI_Wtime()
   endsubroutine ph_tic

   subroutine ph_toc(i)
   !< Stop phase i and accumulate.
   integer(I4P), intent(in) :: i !< Phase.

   if (.not.ph_on) return
   ph_acc(i) = ph_acc(i) + (MPI_Wtime() - ph_t0(i))
   ph_cnt(i) = ph_cnt(i) + 1_I4P
   endsubroutine ph_toc

   subroutine ph_end_step()
   !< Close a step: discard the accumulators at the end of step ADAM_PHASE_SKIP, count the steps after it.
   if (.not.ph_on) return
   ph_it = ph_it + 1_I4P
   if (ph_it == ph_skip) then
      ph_acc = 0._R8P
      ph_cnt = 0_I4P
   elseif (ph_it > ph_skip) then
      ph_steps = ph_steps + 1_I4P
   endif
   endsubroutine ph_end_step

   subroutine ph_report()
   !< Print, on rank 0, per phase: ms per step min / mean / max over the ranks, calls per step, share of the mean step.
   real(R8P)          :: amin(PH_N), amax(PH_N), asum(PH_N) !< Reductions.
   integer            :: np, me, ierr                       !< MPI.
   integer(I4P)       :: i                                  !< Counter.
   real(R8P)          :: per, step_mean                     !< Normalisation.
   logical            :: gradual                            !< Underflow mode in force.
   character(len=200) :: line                               !< Output.

   if (.not.ph_on) return
   call MPI_Comm_size(MPI_COMM_WORLD, np, ierr)
   call MPI_Comm_rank(MPI_COMM_WORLD, me, ierr)
   call MPI_Allreduce(ph_acc, amin, PH_N, MPI_REAL8, MPI_MIN, MPI_COMM_WORLD, ierr)
   call MPI_Allreduce(ph_acc, amax, PH_N, MPI_REAL8, MPI_MAX, MPI_COMM_WORLD, ierr)
   call MPI_Allreduce(ph_acc, asum, PH_N, MPI_REAL8, MPI_SUM, MPI_COMM_WORLD, ierr)
   call ieee_get_underflow_mode(gradual)
   if (me /= 0) return
   per = 1000._R8P / real(max(ph_steps, 1_I4P), R8P)
   step_mean = asum(PH_STEP) / real(np, R8P)
   write(*, '(A,I0,A,I0,A,L1,A,L1)') 'PHASE TIMING: ranks ', np, ', steps timed ', ph_steps, ', barriers skipped ', &
                                     ph_skip_barrier, ', gradual underflow ', gradual
   write(*, '(A)') 'PHASE TIMING: phase          ms/step min      mean       max  calls/step  share of step'
   do i = 1_I4P, PH_N
      write(line, '(A,A14,3F11.3,F12.2,F10.1,A)') 'PHASE TIMING: ', PH_NAME(i), amin(i) * per,                     &
         asum(i) / real(np, R8P) * per, amax(i) * per, real(ph_cnt(i), R8P) / real(max(ph_steps, 1_I4P), R8P), &
         100._R8P * (asum(i) / real(np, R8P)) / max(step_mean, tiny(1._R8P)), ' %'
      write(*, '(A)') trim(line)
   enddo
   endsubroutine ph_report
endmodule adam_phase_timer
